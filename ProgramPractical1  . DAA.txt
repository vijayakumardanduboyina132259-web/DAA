#include <iostream>
#include <vector>
#include <algorithm>
#include <chrono>
#include <cstdlib>

using namespace std;
using namespace chrono;

// Bubble Sort
void bubble(vector<int> &a)
{
    for (int i = 0; i < a.size() - 1; i++)
        for (int j = 0; j < a.size() - i - 1; j++)
            if (a[j] > a[j + 1])
                swap(a[j], a[j + 1]);
}

// Selection Sort
void selection(vector<int> &a)
{
    for (int i = 0; i < a.size() - 1; i++)
    {
        int pos = i;
        for (int j = i + 1; j < a.size(); j++)
            if (a[j] < a[pos])
                pos = j;
        swap(a[i], a[pos]);
    }
}

// Insertion Sort
void insertion(vector<int> &a)
{
    for (int i = 1; i < a.size(); i++)
    {
        int x = a[i];
        int j = i;

        while (j > 0 && a[j - 1] > x)
        {
            a[j] = a[j - 1];
            j--;
        }
        a[j] = x;
    }
}

// Merge Sort
void merge(vector<int> &a, int l, int m, int r)
{
    vector<int> t;
    int i = l, j = m + 1;

    while (i <= m && j <= r)
    {
        if (a[i] < a[j])
            t.push_back(a[i++]);
        else
            t.push_back(a[j++]);
    }

    while (i <= m)
        t.push_back(a[i++]);

    while (j <= r)
        t.push_back(a[j++]);

    for (int k = 0; k < t.size(); k++)
        a[l + k] = t[k];
}

void mergeSort(vector<int> &a, int l, int r)
{
    if (l >= r)
        return;

    int mid = (l + r) / 2;

    mergeSort(a, l, mid);
    mergeSort(a, mid + 1, r);

    merge(a, l, mid, r);
}

// Quick Sort
int divide(vector<int> &a, int low, int high)
{
    int pivot = a[low];
    int i = low + 1;
    int j = high;

    while (true)
    {
        while (i <= high && a[i] <= pivot)
            i++;

        while (a[j] > pivot)
            j--;

        if (i >= j)
            break;

        swap(a[i], a[j]);
    }

    swap(a[low], a[j]);
    return j;
}

void quickSort(vector<int> &a, int low, int high)
{
    if (low < high)
    {
        int p = divide(a, low, high);
        quickSort(a, low, p - 1);
        quickSort(a, p + 1, high);
    }
}

{
    const int n = 100;
    vector<int> data(n);

    srand(time(NULL));

    for (int i = 0; i < n; i++)
        data[i] = rand() % 1000;

    cout << "Number of Elements = " << n << "\n\n";

    vector<int> temp;

    auto test = [&](string name, void (*sortFunc)(vector<int>&))
    {
        temp = data;
        auto start = high_resolution_clock::now();
        sortFunc(temp);
        auto stop = high_resolution_clock::now();

        cout << name << " : "
             << duration_cast<microseconds>(stop - start).count()
             << " microseconds\n";
    };

    test("Bubble Sort Time   ", bubble);
    test("Selection Sort Time", selection);
    test("Insertion Sort Time", insertion);

    temp = data;
    auto start = high_resolution_clock::now();
    mergeSort(temp, 0, n - 1);
    auto stop = high_resolution_clock::now();
    cout << "Merge Sort Time    : "

         << " microseconds\n";

    temp = data;
    start = high_resolution_clock::now();
    quickSort(temp, 0, n - 1);
    stop = high_resolution_clock::now();
    cout << "Quick Sort Time    : "

         << " microseconds\n";

    return 0;
}
