#include <iostream>
#include <vector>
#include <chrono>

using namespace std;
using namespace chrono;

// Linear Search
int linearSearch(const vector<int>& data, int target)
{
    int i = 0;
    while (i < data.size())
    {
        if (data[i] == target)
            return i;
        i++;
    }
    return -1;
}

// Binary Search
int binarySearch(const vector<int>& data, int target)
{
    int left = 0;
    int right = data.size() - 1;

    while (left <= right)
    {
        int middle = (left + right) / 2;

        if (data[middle] == target)
            return middle;

        if (target > data[middle])
            left = middle + 1;
        else
            right = middle - 1;
    }

    return -1;
}

int main()
{
    const int SIZE = 100000;
    vector<int> numbers;

    // Fill the vector
    for (int i = 1; i <= SIZE; i++)
        numbers.push_back(i);

    int target;
    cout << "Enter element to search: ";
    cin >> target;

    auto begin = high_resolution_clock::now();
    int result1 = linearSearch(numbers, target);
    auto end = high_resolution_clock::now();

    cout << "\nLinear Search\n";
    if (result1 >= 0)
        cout << "Element found at index " << result1 << endl;
    else
        cout << "Element not found" << endl;

    cout << "Time Taken : "
         << duration_cast<microseconds>(end - begin).count()
         << " microseconds" << endl;

    begin = high_resolution_clock::now();
    int result2 = binarySearch(numbers, target);
    end = high_resolution_clock::now();

    cout << "\nBinary Search\n";
    if (result2 >= 0)
        cout << "Element found at index " << result2 << endl;
    else
        cout << "Element not found" << endl;

    cout << "Time Taken : "
         << duration_cast<microseconds>(end - begin).count()
         << " microseconds" << endl;

    return 0;
}
